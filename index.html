from docx import Document from docx.shared import Inches from docx.enum.text import WD_PARAGRAPH_ALIGNMENT import matplotlib.pyplot as plt dates = ["4 May 2026", "5 May 2026", "6 May 2026", "7 May 2026"] pax = [870, 848, 870, 772] chicken = [525, 490, 490, 420] beef = [300, 360, 360, 420] # Protein comparison graph plt.figure(figsize=(8,5)) x = range(len(dates)) width = 0.35 plt.bar([i - width/2 for i in x], chicken, width=width) plt.bar([i + width/2 for i in x], beef, width=width) plt.xticks(list(x), dates, rotation=10) plt.ylabel("Protein Portions") plt.title("Chicken vs Beef Portions") protein_chart = "/mnt/data/protein_chart.png" plt.savefig(protein_chart, bbox_inches='tight') plt.close() # Coverage graph total_protein = [c+b for c,b in zip(chicken, beef)] plt.figure(figsize=(8,5)) plt.bar(dates, pax) plt.plot(dates, total_protein, marker='o') plt.ylabel("Count") plt.title("Pax Served vs Total Protein Portions") coverage_chart = "/mnt/data/coverage_chart.png" plt.savefig(coverage_chart, bbox_inches='tight') plt.close() doc = Document() title = doc.add_heading("KITCHEN USAGE OBSERVATION & RECOMMENDATION REPORT", level=1) title.alignment = WD_PARAGRAPH_ALIGNMENT.CENTER doc.add_paragraph("Period Reviewed: 4 May 2026 – 7 May 2026") doc.add_heading("Purpose", level=2) doc.add_paragraph( "The purpose of this report is to review food usage against pax served and identify areas where production can be improved to reduce shortages and wastage." ) doc.add_heading("1. Protein Production Observation", level=2) doc.add_paragraph( "Our observation is that pax choose either chicken or beef during service. Chicken demand was generally higher than beef demand." ) doc.add_paragraph("Average customer preference observed:") doc.add_paragraph("- Chicken: approximately 57%") doc.add_paragraph("- Beef: approximately 43%") doc.add_picture(protein_chart, width=Inches(6.5)) doc.add_heading("2. Best Controlled Production", level=2) doc.add_paragraph( "5 May 2026 had the best production control with only 2 extra portions above pax served. This indicates very low wastage and good planning." ) doc.add_heading("3. Overproduction and Shortages", level=2) doc.add_paragraph( "On 7 May 2026 there was slight overproduction with approximately 68 extra protein portions." ) doc.add_paragraph( "On 4 May 2026 and 6 May 2026 there were estimated shortages of approximately 45 portions and 20 portions respectively." ) doc.add_picture(coverage_chart, width=Inches(6.5)) doc.add_heading("4. Rice and Nsima Observation", level=2) doc.add_paragraph( "Rice production appeared generally high while nsima production changed depending on demand." ) doc.add_paragraph( "Because service is very busy, it is difficult to record every customer choice between rice and nsima." ) doc.add_heading("5. Maintain a Small Safety Buffer", level=2) doc.add_paragraph( "We recommend preparing approximately 5% above expected pax numbers to avoid shortages while also controlling wastage." ) doc.add_paragraph("Example for 100 Pax") table = doc.add_table(rows=1, cols=3) hdr = table.rows[0].cells hdr[0].text = "Item" hdr[1].text = "Expected Demand" hdr[2].text = "Recommended Production" example_rows = [ ("Chicken", "57 portions", "60 portions (8–9 chickens)"), ("Beef", "43 portions", "45 portions (7–8kg beef)"), ] for item, demand, production in example_rows: row = table.add_row().cells row[0].text = item row[1].text = demand row[2].text = production doc.add_heading("6. Recommended Average Planning per 100 Pax", level=2) table2 = doc.add_table(rows=1, cols=2) hdr2 = table2.rows[0].cells hdr2[0].text = "Item" hdr2[1].text = "Recommended Quantity" items = [ ("Chicken", "8 chickens"), ("Beef", "7kg"), ("Rice", "14kg–15kg"), ("Ufa/Nsima", "10kg"), ("Cooking Oil", "2–3 litres"), ("Tomatoes", "3–4kg"), ("Butter", "0.4kg"), ("Baking Flour", "3–4kg"), ("Table Salt", "0.7kg"), ("Onions", "0.5kg"), ("Cabbage", "5 heads"), ("Frooty", "2 bottles"), ] for item, qty in items: row = table2.add_row().cells row[0].text = item row[1].text = qty doc.add_heading("7. Recommendations", level=2) doc.add_paragraph("- Continue using average consumption trends for planning.") doc.add_paragraph("- Maintain approximately 5% extra production as a safety buffer.") doc.add_paragraph("- Monitor major leftovers such as rice, nsima, and protein portions.") doc.add_paragraph("- Keep chicken production slightly higher than beef production.") doc.add_heading("Conclusion", level=2) doc.add_paragraph( "Kitchen production is generally close to operational requirements. Using average consumption trends together with daily leftover observations will help improve food planning, stock control, cost management, and reduction of wastage." ) output_path = "/mnt/data/Kitchen_Usage_Observation_Report.docx" doc.save(output_path) print(f"Saved report to {output_path}")
